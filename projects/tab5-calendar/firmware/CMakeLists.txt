cmake_minimum_required(VERSION 3.16)
set(IDF_TARGET "esp32p4")
set(PROJECT_VER "0.1.2-dev")
set(TAB5_COMPONENTS "${CMAKE_CURRENT_LIST_DIR}/../../tab5-console/components")
set(EXTRA_COMPONENT_DIRS "${TAB5_COMPONENTS}/tab5_hal" "${TAB5_COMPONENTS}/tab5_app_boot" "${TAB5_COMPONENTS}/tab5_radio" "${TAB5_COMPONENTS}/esp_lcd_st7121" "${TAB5_COMPONENTS}/fatfs")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(Tab5Calendar)
add_custom_target(tab5_calendar_sd_only COMMAND ${CMAKE_COMMAND} -E echo "SD application: install using Console" COMMAND ${CMAKE_COMMAND} -E false)
foreach(flash_target flash app-flash bootloader-flash partition-table-flash)
    if(TARGET ${flash_target})
        add_dependencies(${flash_target} tab5_calendar_sd_only)
    endif()
endforeach()
